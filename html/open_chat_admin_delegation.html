クラブ管理人・副管理人
権限委譲システム

オープンチャット（クラブ）における管理人の譲渡、副管理人の任命・解任、および管理人退室時のフロントエンド主導型移行フローの設計ドキュメントです。
自動フォールバックは行わず、退室前に必ずUIで移行先を選択させる安全設計を採用しています。

設計の核心: role カラムの3値化

従来の owner_id カラムと role='admin' の併用は判定ロジックが二重になりバグの元。
members.role を3値に統一し、ロール判定を1箇所に一本化します。

副管理人の制限: 副管理人(admin)はオーナー(owner)および他の副管理人に対してキック/BANを実行できません。RPC側でガードします。

データ整合性: UNIQUE部分インデックス

各ルームにオーナーは必ず1名だけ存在することをDBレベルで保証します。 

CREATE UNIQUE INDEX idx_open_chat_one_owner_per_room
ON open_chat_members (room_id)
WHERE role = 'owner';

※ この部分インデックスにより、同じ room_id で role='owner' のレコードは1つしか挿入できなくなります。

ロール判定ロジック（Flutter側）— 1箇所に一本化

管理人 = myMember.role == 'owner'
副管理人 = myMember.role == 'admin'
一般 = myMember.role == 'member'
✅ room.ownerId の参照は不要。myMember.role だけで完結

RPC関数設計

新規作成3つ + 既存改修2つ。すべて SECURITY DEFINER + auth.uid() チェック付き。

transfer_room_ownership 新規
p_room_id UUID p_new_owner_id UUID → JSONB

管理人権限を別のメンバーに譲渡する。旧オーナーは admin に降格、新オーナーは owner に昇格。

CREATE OR REPLACE FUNCTION public.transfer_room_ownership(
  p_room_id UUID, p_new_owner_id UUID
) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_caller_id UUID := auth.uid();
  v_is_owner BOOLEAN;
BEGIN
  -- 呼び出し元がオーナーか確認
  SELECT EXISTS(
    SELECT 1 FROM open_chat_members
    WHERE room_id = p_room_id AND user_id = v_caller_id AND role = 'owner'
  ) INTO v_is_owner;

  IF NOT v_is_owner THEN
    RETURN jsonb_build_object('success', false, 'error', '管理人のみが権限を譲渡できます');
  END IF;

  -- 新しいオーナーがメンバーか確認
  IF NOT EXISTS(
    SELECT 1 FROM open_chat_members
    WHERE room_id = p_room_id AND user_id = p_new_owner_id
  ) THEN
    RETURN jsonb_build_object('success', false, 'error', 'メンバーではないユーザーには譲渡できません');
  END IF;

  -- 旧オーナーを admin に降格
  UPDATE open_chat_members SET role = 'admin'
  WHERE room_id = p_room_id AND user_id = v_caller_id;

  -- 新オーナーを owner に昇格
  UPDATE open_chat_members SET role = 'owner'
  WHERE room_id = p_room_id AND user_id = p_new_owner_id;

  RETURN jsonb_build_object('success', true);
END; $$;
assign_admin_role 新規
p_room_id UUID p_target_user_id UUID → JSONB

一般メンバーを副管理人に任命する。オーナー(role='owner')のみ呼び出し可能。

CREATE OR REPLACE FUNCTION public.assign_admin_role(
  p_room_id UUID, p_target_user_id UUID
) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE v_caller_id UUID := auth.uid();
BEGIN
  IF NOT EXISTS(
    SELECT 1 FROM open_chat_members
    WHERE room_id = p_room_id AND user_id = v_caller_id AND role = 'owner'
  ) THEN
    RETURN jsonb_build_object('success', false, 'error', '管理人のみが副管理人を任命できます');
  END IF;

  UPDATE open_chat_members SET role = 'admin'
  WHERE room_id = p_room_id AND user_id = p_target_user_id AND role = 'member';

  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'error', '対象はこのルームの一般メンバーではありません');
  END IF;

  RETURN jsonb_build_object('success', true);
END; $$;
revoke_admin_role 新規
p_room_id UUID p_target_user_id UUID → JSONB

副管理人を一般メンバーに降格する。オーナー(owner)自身は解任不可。

CREATE OR REPLACE FUNCTION public.revoke_admin_role(
  p_room_id UUID, p_target_user_id UUID
) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE v_caller_id UUID := auth.uid();
BEGIN
  IF NOT EXISTS(
    SELECT 1 FROM open_chat_members
    WHERE room_id = p_room_id AND user_id = v_caller_id AND role = 'owner'
  ) THEN
    RETURN jsonb_build_object('success', false, 'error', '管理人のみが副管理人を解任できます');
  END IF;

  -- オーナー(owner)は解任できない
  IF EXISTS(
    SELECT 1 FROM open_chat_members
    WHERE room_id = p_room_id AND user_id = p_target_user_id AND role = 'owner'
  ) THEN
    RETURN jsonb_build_object('success', false, 'error', '管理人を副管理人から外すことはできません');
  END IF;

  UPDATE open_chat_members SET role = 'member'
  WHERE room_id = p_room_id AND user_id = p_target_user_id AND role = 'admin';

  RETURN jsonb_build_object('success', true);
END; $$;
create_open_chat_room 改修

ルーム作成時にオーナーを role='owner' で登録するように変更。
role='admin' → role='owner' 

-- 変更前
VALUES (v_room_id, v_user_id, 'admin');

-- 変更後
VALUES (v_room_id, v_user_id, 'owner');
leave_open_chat_room 既存・変更なし

自動フォールバックロジックは入れない。
単にメンバーレコードを削除するだけの既存動作を維持。
オーナーがこれを呼ぶ前に、フロントエンドで必ず transfer_room_ownership を完了させる前提。
RLSでオーナーも呼び出せるよう調整のみ。

マイグレーションSQL

既存データを変換しながらスキーマを変更する手順。必ずバックアップ後に実行してください。

-- ========================================
-- Step 1: CHECK制約の変更
-- ========================================
ALTER TABLE open_chat_members
  DROP CONSTRAINT IF EXISTS open_chat_members_role_check;

ALTER TABLE open_chat_members
  ADD CONSTRAINT open_chat_members_role_check
  CHECK (role IN ('owner', 'admin', 'member'));

-- ========================================
-- Step 2: 既存オーナーの role を 'admin' → 'owner' に変換
-- open_chat_rooms.owner_id を参照して特定
-- ========================================
UPDATE open_chat_members m
SET role = 'owner'
FROM open_chat_rooms r
WHERE m.room_id = r.id
  AND m.user_id = r.owner_id
  AND m.role = 'admin';

-- ========================================
-- Step 3: owner_id カラムの削除
-- ========================================
ALTER TABLE open_chat_rooms
  DROP COLUMN IF EXISTS owner_id;

-- ========================================
-- Step 4: ユニーク部分インデックス作成
-- ========================================
CREATE UNIQUE INDEX idx_open_chat_one_owner_per_room
ON open_chat_members (room_id)
WHERE role = 'owner';

-- ========================================
-- Step 5: RLSポリシーの更新
-- 旧: auth.uid() = owner_id
-- 新: members.role = 'owner'
-- ========================================
DROP POLICY IF EXISTS "ルーム管理者は更新可能" ON open_chat_rooms;
DROP POLICY IF EXISTS "ルーム管理者は削除可能" ON open_chat_rooms;

CREATE POLICY "ルーム管理者は更新可能" ON open_chat_rooms
  FOR UPDATE USING (
    EXISTS (
      SELECT 1 FROM open_chat_members
      WHERE room_id = open_chat_rooms.id
        AND user_id = auth.uid()
        AND role = 'owner'
    )
  );

CREATE POLICY "ルーム管理者は削除可能" ON open_chat_rooms
  FOR DELETE USING (
    EXISTS (
      SELECT 1 FROM open_chat_members
      WHERE room_id = open_chat_rooms.id
        AND user_id = auth.uid()
        AND role = 'owner'
    )
  );
注意点
  • Step 2 を先に実行しないと owner_id 削除後にオーナー特定不能
  • 既存の副管理人(role='admin')はそのまま維持される
  • owner_id を参照している他RPC/ビューも同時に修正が必要
影響範囲の確認コマンド
-- owner_id を参照しているオブジェクト一覧
SELECT proname FROM pg_proc
WHERE prosrc LIKE '%owner_id%';

副管理人からの見え方

副管理人(admin)のメンバー一覧では、「強制退会」ボタンのみを表示。
「副管理人に任命」「副管理人を外す」「管理人を譲る」は表示しない。
また、オーナー(owner)および他の副管理人(admin)に対する「強制退会」ボタンも非表示にする。

OpenChatActionNotifier に追加する3メソッド + ロール判定ヘルパー

ロール判定ヘルパー（OpenChatMember に拡張メソッドとして追加）
extension OpenChatMemberRoleX on OpenChatMember {
  bool get isOwner  => role == 'owner';
  bool get isAdmin  => role == 'admin';
  bool get isMember => role == 'member';
  // 管理人 or 副管理人（モデレーション権限あり）
  bool get isModerator => isOwner || isAdmin;
}
/// 管理人権限を譲渡する
Future<String?> transferOwnership(String roomId, String newOwnerId) async {
  final supabase = ref.read(supabaseProvider);
  final response = await supabase.rpc('transfer_room_ownership', params: {
    'p_room_id': roomId,
    'p_new_owner_id': newOwnerId,
  });
  if (response['success'] == true) {
    ref.invalidate(openChatMyMemberProvider(roomId));
    ref.invalidate(openChatMembersProvider(roomId));
    return null;
  }
  return response['error'] ?? '譲渡に失敗しました';
}

/// 副管理人を任命する
Future<String?> assignAdmin(String roomId, String targetUserId) async {
  final supabase = ref.read(supabaseProvider);
  final response = await supabase.rpc('assign_admin_role', params: {
    'p_room_id': roomId,
    'p_target_user_id': targetUserId,
  });
  if (response['success'] == true) {
    ref.invalidate(openChatMembersProvider(roomId));
    return null;
  }
  return response['error'] ?? '任命に失敗しました';
}

/// 副管理人を解任する
Future<String?> revokeAdmin(String roomId, String targetUserId) async {
  final supabase = ref.read(supabaseProvider);
  final response = await supabase.rpc('revoke_admin_role', params: {
    'p_room_id': roomId,
    'p_target_user_id': targetUserId,
  });
  if (response['success'] == true) {
    ref.invalidate(openChatMembersProvider(roomId));
    return null;
  }
  return response['error'] ?? '解任に失敗しました';
}
